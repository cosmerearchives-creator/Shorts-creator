cmake_minimum_required(VERSION 3.22.1)
project(shorts_speech LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(WHISPER_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/whisper.cpp")
if(NOT EXISTS "${WHISPER_ROOT}/CMakeLists.txt")
  message(FATAL_ERROR "Missing vendor/whisper.cpp; fetch pinned source before compiling")
endif()
add_subdirectory("${WHISPER_ROOT}" "${CMAKE_CURRENT_BINARY_DIR}/whisper_build")
add_library(shorts_speech SHARED whisper_jni.cpp)
target_include_directories(shorts_speech PRIVATE "${WHISPER_ROOT}/include")
target_link_libraries(shorts_speech PRIVATE whisper log)
